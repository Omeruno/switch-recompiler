cmake_minimum_required(VERSION 3.19)
project(switch_visual_browser LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT LITEHTML_ROOT_DIR)
  message(FATAL_ERROR "Set LITEHTML_ROOT_DIR to the pinned renderer")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/litehtml.lock.json" lock)
string(REGEX MATCH "\"commit\": \"([a-f0-9]+)\"" unused "${lock}")
set(expected "${CMAKE_MATCH_1}")
execute_process(COMMAND git -C ${LITEHTML_ROOT_DIR} rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "Renderer revision differs from litehtml.lock.json")
endif()
set(LITEHTML_ENABLE_LINT OFF CACHE BOOL "" FORCE)
set(LITEHTML_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
add_subdirectory(${LITEHTML_ROOT_DIR} litehtml)
add_library(visual_profile STATIC ../source/profile.c ../source/ca_bundle.c)
add_library(visual_core STATIC source/container.cpp source/resources.cpp)
target_include_directories(visual_profile PUBLIC ../source)
target_include_directories(visual_core PUBLIC source ../source)
target_compile_options(visual_core PRIVATE -Wall -Wextra)
if(SWITCH_VISUAL_TARGET)
  include_directories($ENV{DEVKITPRO}/portlibs/switch/include/SDL2)
  target_include_directories(visual_core PRIVATE $ENV{DEVKITPRO}/portlibs/switch/include/SDL2 $ENV{DEVKITPRO}/portlibs/switch/include)
  target_include_directories(visual_profile PRIVATE $ENV{DEVKITPRO}/portlibs/switch/include)
  target_link_libraries(visual_core PUBLIC litehtml visual_profile)
else()
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf SDL2_image)
  pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
  target_link_libraries(visual_profile PUBLIC PkgConfig::CURL)
  target_link_libraries(visual_core PUBLIC litehtml visual_profile PkgConfig::SDL)
  add_executable(visual-host source/main.cpp)
  target_link_libraries(visual-host PRIVATE visual_core)
  add_executable(visual-input-test tests/input_test.cpp)
  target_include_directories(visual-input-test PRIVATE source)
  target_compile_options(visual-input-test PRIVATE -Wall -Wextra -Werror)
  enable_testing()
  add_test(NAME touch-state COMMAND visual-input-test)
  add_test(NAME visual-layout COMMAND visual-host ${CMAKE_CURRENT_SOURCE_DIR}/romfs ${CMAKE_CURRENT_SOURCE_DIR}/../romfs/DejaVuSans.ttf --test ${CMAKE_CURRENT_BINARY_DIR}/visual-preview.bmp)
  set_tests_properties(visual-layout PROPERTIES ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
