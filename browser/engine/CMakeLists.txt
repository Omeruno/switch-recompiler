cmake_minimum_required(VERSION 3.19)
project(switch_engine_probe C)
set(CMAKE_C_STANDARD 11)
if(NOT WAMR_ROOT_DIR)
  message(FATAL_ERROR "Set WAMR_ROOT_DIR to the pinned dependency")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/wamr.lock.json" lock)
string(REGEX MATCH "\"commit\": \"([a-f0-9]+)\"" unused "${lock}")
set(expected_commit "${CMAKE_MATCH_1}")
execute_process(COMMAND git -C ${WAMR_ROOT_DIR} rev-parse HEAD OUTPUT_VARIABLE actual_commit OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual_commit STREQUAL expected_commit)
  message(FATAL_ERROR "Runtime revision differs from wamr.lock.json")
endif()
add_compile_definitions(ENGINE_WAMR_REVISION="${expected_commit}")
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 0)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_FAST_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_SIMD 0)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_SHARED_MEMORY 1)
set(WAMR_BUILD_EXCE_HANDLING 1)
set(WAMR_BUILD_EXTENDED_CONST_EXPR 1)
set(WAMR_BUILD_BULK_MEMORY 1)
set(WAMR_BUILD_BULK_MEMORY_OPT 1)
set(WAMR_BUILD_CALL_INDIRECT_OVERLONG 1)
set(WAMR_BUILD_THREAD_MGR 0)
set(WAMR_DISABLE_HW_BOUND_CHECK 1)
set(WAMR_DISABLE_WAKEUP_BLOCKING_OP 1)
set(WAMR_BUILD_MINI_LOADER 0)
if(SWITCH_ENGINE_TARGET)
  set(WAMR_BUILD_TARGET AARCH64)
  set(WAMR_BUILD_PLATFORM switch)
  set(SHARED_PLATFORM_CONFIG ${CMAKE_CURRENT_SOURCE_DIR}/platform/switch/shared_platform.cmake)
else()
  set(WAMR_BUILD_PLATFORM linux)
endif()
include(${WAMR_ROOT_DIR}/build-scripts/runtime_lib.cmake)
add_library(engine_runtime STATIC ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(engine_runtime PUBLIC ${WAMR_ROOT_DIR}/core/iwasm/include)
add_library(engine_probe STATIC src/probe.c src/fixtures.c)
target_link_libraries(engine_probe PUBLIC engine_runtime)
target_include_directories(engine_probe PUBLIC src ${SHARED_DIR}/platform/include)
if(SWITCH_ENGINE_TARGET)
  target_compile_definitions(engine_runtime PUBLIC __SWITCH__)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(engine_runtime PUBLIC m dl Threads::Threads)
  add_executable(engine-host src/host_main.c)
  target_link_libraries(engine-host PRIVATE engine_probe)
  enable_testing()
  add_test(NAME engine-features COMMAND engine-host)
endif()
